// Retorna a data atual real do sistema no formato YYYY-MM-DD
const obterDataHojeISO = () => {
  const hoje = new Date();
  const ano = hoje.getFullYear();
  const mes = String(hoje.getMonth() + 1).padStart(2, '0');
  const dia = String(hoje.getDate()).padStart(2, '0');
  return `${ano}-${mes}-${dia}`;
};

// Gera um apontamento diário reiniciado (limpo para um novo dia)
const gerarPlanilhaReiniciada = (listaFuncionarios, modo = 'PADRAO') => {
  const novoDia = {};
  listaFuncionarios.forEach((emp) => {
    novoDia[emp.id] = {
      status: modo === 'EM_BRANCO' ? 'NÃO CONFIRMADO' : 'PRESENTE',
      frente: emp.frentePadrao,
      subfrente: emp.subfrentePadrao || '',
      responsavel: emp.responsavel || ''
    };
  });
  return novoDia;
};

    window.addEventListener('focus', verificarViradaDeDia);
    return () => window.removeEventListener('focus', verificarViradaDeDia);
  }, [dataHojeISO]);

  // ==========================================
  // 2. AUTO-SAVE TODA VEZ QUE EDITAR O SITE
  // ==========================================
  useEffect(() => {
    if (!dadosCarregados) return;
    if (ignorarProximoSaveRef.current) {
      ignorarProximoSaveRef.current = false;
      return;
    }

    setStatusSave('Salvando alterações...');
    const timer = setTimeout(async () => {
      const agora = new Date().toLocaleTimeString('pt-BR', {
        hour: '2-digit',
        minute: '2-digit',
        second: '2-digit'
      });

      if (user && db) {
        try {
          const stateDocRef = doc(db, 'artifacts', appId, 'users', user.uid, 'configuracoes', 'estado_geral');
          await setDoc(stateDocRef, {
            funcionariosJson: JSON.stringify(funcionarios),
            frentesConfigJson: JSON.stringify(frentesConfig),
            apontamentosJson: JSON.stringify(apontamentosPorDia),
            updatedAt: agora
          });
          setUltimoSaveHora(agora);
          setStatusSave('Salvo automaticamente');
        } catch (err) {
          console.error('Erro no auto-save:', err);
          setStatusSave('Erro ao salvar na nuvem');
        }
      } else {
        setUltimoSaveHora(agora);
        setStatusSave('Salvo na sessão');
      }
    }, 600);

    return () => clearTimeout(timer);
  }, [funcionarios, frentesConfig, apontamentosPorDia, user, dadosCarregados]);

  // Chave do Dia e Turno Ativo
  const chaveAtual = `${dataSelecionada}_${turnoSelecionado}`;

  // A cada novo dia (ou data ainda não aberta), cria automaticamente uma planilha reiniciada
  // para Manhã e Tarde sem alterar nenhuma planilha dos dias anteriores
  useEffect(() => {
    if (!dadosCarregados) return;
    const chaveManha = `${dataSelecionada}_MANHA`;
    const chaveTarde = `${dataSelecionada}_TARDE`;

    if (!apontamentosPorDia[chaveManha] || !apontamentosPorDia[chaveTarde]) {
      setApontamentosPorDia((prev) => {
        const novoEstado = { ...prev };
        if (!novoEstado[chaveManha]) {
          novoEstado[chaveManha] = gerarPlanilhaReiniciada(funcionarios, 'PADRAO');
        }
        if (!novoEstado[chaveTarde]) {
          novoEstado[chaveTarde] = gerarPlanilhaReiniciada(funcionarios, 'PADRAO');
        }
        return novoEstado;
      });
    }
  }, [dataSelecionada, apontamentosPorDia, funcionarios, dadosCarregados]);

  // Lista de todas as datas que possuem planilhas salvas no histórico (ordenadas da mais recente para a mais antiga)
  const datasSalvasNoHistorico = useMemo(() => {
    const conjuntoDatas = new Set([dataHojeISO, '2026-11-11', dataSelecionada]);
    Object.keys(apontamentosPorDia).forEach((chave) => {
      const [dt] = chave.split('_');
      if (dt) conjuntoDatas.add(dt);
    });
    return Array.from(conjuntoDatas).sort((a, b) => b.localeCompare(a));
  }, [apontamentosPorDia, dataHojeISO, dataSelecionada]);

  // Navegar para o dia anterior ou próximo dia facilmente
  const alterarDiaOffset = (diasOffset) => {
    const [ano, mes, dia] = dataSelecionada.split('-').map(Number);
    const dt = new Date(ano, mes - 1, dia);
    dt.setDate(dt.getDate() + diasOffset);
    const novoAno = dt.getFullYear();
    const novoMes = String(dt.getMonth() + 1).padStart(2, '0');
    const novoDia = String(dt.getDate()).padStart(2, '0');
    setDataSelecionada(`${novoAno}-${novoMes}-${novoDia}`);
  };

  // Reiniciar manualmente a planilha do dia atual sem afetar os dias anteriores
  const reiniciarPlanilhaDiaAtual = (modo = 'PADRAO') => {
    setApontamentosPorDia((prev) => ({
      ...prev,
      [chaveAtual]: gerarPlanilhaReiniciada(funcionarios, modo)
    }));
    setModalReiniciarDia(false);
    mostrarAviso(
      modo === 'PADRAO'
        ? `Planilha de ${dataSelecionada.split('-').reverse().join('/')} (${turnoSelecionado}) reiniciada com a alocação padrão!`
        : `Planilha de ${dataSelecionada.split('-').reverse().join('/')} (${turnoSelecionado}) zerada para nova chamada!`
    );
  };

  const apontamentoAtual = apontamentosPorDia[chaveAtual] || {};

  // Atualiza qualquer campo de apontamento de um colaborador (e dispara o auto-save)
  const atualizarApontamento = (id, camposObj) => {
    setApontamentosPorDia((prev) => {
      const atualEmp = prev[chaveAtual]?.[id] || {};
      return {
        ...prev,
        [chaveAtual]: {
          ...(prev[chaveAtual] || {}),
          [id]: {
            ...atualEmp,
            ...camposObj
          }
        }
      };
    });
  };

  // ==========================================
  // PERSONALIZAÇÃO DE FUNCIONÁRIOS EM TEMPO REAL
  // ==========================================
  const atualizarCadastroFuncionario = (idAtual, novosCampos, sincronizarTurnoAtual = true) => {
    setFuncionarios((prev) =>
      prev.map((emp) => (emp.id === idAtual ? { ...emp, ...novosCampos } : emp))
    );

    if (sincronizarTurnoAtual) {
      const camposTurno = {};
      if (novosCampos.frentePadrao !== undefined) camposTurno.frente = novosCampos.frentePadrao;
      if (novosCampos.subfrentePadrao !== undefined) camposTurno.subfrente = novosCampos.subfrentePadrao;
      if (novosCampos.responsavel !== undefined) camposTurno.responsavel = novosCampos.responsavel;
      if (Object.keys(camposTurno).length > 0) {
        atualizarApontamento(idAtual, camposTurno);
      }
    }
  };

  // Altera o ID (Matrícula) do colaborador migrando também o histórico de dias
  const alterarIdFuncionario = (idAntigo, novoIdBruto) => {
    const novoId = (novoIdBruto || '').toUpperCase().trim();
    if (!novoId || novoId === idAntigo) return false;

    const jaExiste = funcionarios.some((f) => f.id === novoId);
    if (jaExiste) {
      mostrarAviso(`O ID "${novoId}" já está sendo usado por outro colaborador!`);
      return false;
    }

    setFuncionarios((prev) =>
      prev.map((emp) => (emp.id === idAntigo ? { ...emp, id: novoId } : emp))
    );

    setApontamentosPorDia((prev) => {
      const novoMapa = {};
      Object.keys(prev).forEach((chaveDia) => {
        const mapaDia = { ...prev[chaveDia] };
        if (mapaDia[idAntigo]) {
          mapaDia[novoId] = mapaDia[idAntigo];
          delete mapaDia[idAntigo];
        }
        novoMapa[chaveDia] = mapaDia;
      });
      return novoMapa;
    });

    if (colabEditandoRapido?.id === idAntigo) {
      setColabEditandoRapido((prev) => (prev ? { ...prev, id: novoId } : null));
    }
    setIdTempModal(novoId);
    mostrarAviso(`ID alterado de ${idAntigo} para ${novoId} em tempo real!`);
    return true;
  };

  // Transferir colaborador de setor (CIVIL <-> METÁLICA) em tempo real
  const alterarSetorFuncionario = (idAlvo, novoSetor) => {
    const primeiraFrente = frentesConfig[novoSetor]?.[0];
    const nomeFrente = primeiraFrente ? primeiraFrente.nome : '';
    const subPadrao =
      primeiraFrente && primeiraFrente.temSubfrentes
        ? primeiraFrente.subfrentes[0] || ''
        : '';

    atualizarCadastroFuncionario(
      idAlvo,
      {
        setor: novoSetor,
        frentePadrao: nomeFrente,
        subfrentePadrao: subPadrao,
        responsavel: novoSetor === 'CIVIL' ? 'CIVIL' : 'METÁLICA'
      },
      true
    );
  };

  // Excluir colaborador em tempo real
  const excluirFuncionario = (idAlvo) => {
    const alvo = funcionarios.find((f) => f.id === idAlvo);
    setFuncionarios((prev) => prev.filter((emp) => emp.id !== idAlvo));
    if (colabEditandoRapido?.id === idAlvo) {
      setColabEditandoRapido(null);
    }
    setConfirmarExclusaoId(null);
    if (alvo) {
      mostrarAviso(`Colaborador ${alvo.nome} (${alvo.id}) excluído!`);
    }
  };

  // Lista enriquecida com os dados do turno atual
  const listaCompletaTurno = useMemo(() => {
    return funcionarios.map((emp) => {
      const reg = apontamentoAtual[emp.id] || {
        status: 'PRESENTE',
        frente: emp.frentePadrao,
        subfrente: emp.subfrentePadrao || '',
        responsavel: emp.responsavel
      };
      return {
        ...emp,
        statusAtual: reg.status || 'PRESENTE',
        frenteAtual: reg.frente || emp.frentePadrao,
        subfrenteAtual: reg.subfrente !== undefined ? reg.subfrente : emp.subfrentePadrao || '',
        responsavelAtual: reg.responsavel !== undefined ? reg.responsavel : emp.responsavel || ''
      };
    });
  }, [funcionarios, apontamentoAtual]);

  // Frentes do Setor Selecionado
  const frentesDoSetorAtivo = useMemo(() => {
    return frentesConfig[setorFiltro] || [];
  }, [frentesConfig, setorFiltro]);

  // Encontrar objeto de configuração de uma frente pelo nome e setor
  const obterConfigFrente = (setor, nomeFrente) => {
    const lista = frentesConfig[setor] || [];
    return lista.find((f) => f.nome === nomeFrente);
  };

  // Agrupamento exato para o Esqueleto da Planilha (Colunas lado a lado)
  const dadosEsqueletoPlanilha = useMemo(() => {
    const colunas = frentesDoSetorAtivo.map((frenteObj) => {
      const colabs = listaCompletaTurno
        .filter(
          (emp) =>
            emp.setor === setorFiltro &&
            emp.statusAtual === 'PRESENTE' &&
            emp.frenteAtual === frenteObj.nome
        )
        .sort((a, b) => {
          if (frenteObj.temSubfrentes) {
            const cmpSub = (a.subfrenteAtual || '').localeCompare(b.subfrenteAtual || '');
            if (cmpSub !== 0) return cmpSub;
          }
          return a.id.localeCompare(b.id);
        });

      return {
        ...frenteObj,
        colaboradores: colabs
      };
    });

    const maxLinhas = Math.max(...colunas.map((c) => c.colaboradores.length), 15);
    return { colunas, maxLinhas };
  }, [frentesDoSetorAtivo, listaCompletaTurno, setorFiltro]);

  // Filtragem para Modo Celular e Lista de Apontamento
  const listaFiltrada = useMemo(() => {
    return listaCompletaTurno.filter((item) => {
      if (item.setor !== setorFiltro) return false;
      if (frenteFiltro !== 'TODAS' && item.frenteAtual !== frenteFiltro) return false;
      if (statusFiltro !== 'TODOS') {
        if (statusFiltro === 'OCORRENCIAS' && item.statusAtual === 'PRESENTE') return false;
        if (statusFiltro !== 'OCORRENCIAS' && item.statusAtual !== statusFiltro) return false;
      }
      if (busca.trim() !== '') {
        const t = busca.toLowerCase();
        return (
          item.nome.toLowerCase().includes(t) ||
          item.id.toLowerCase().includes(t) ||
          item.frenteAtual.toLowerCase().includes(t) ||
          (item.subfrenteAtual && item.subfrenteAtual.toLowerCase().includes(t))
        );
      }
      return true;
    });
  }, [listaCompletaTurno, setorFiltro, frenteFiltro, statusFiltro, busca]);

  // KPIs do Setor Atual
  const kpis = useMemo(() => {
    const baseSetor = listaCompletaTurno.filter((i) => i.setor === setorFiltro);
    const presentes = baseSetor.filter((i) => i.statusAtual === 'PRESENTE').length;
    const faltas = baseSetor.filter((i) => i.statusAtual === 'FALTA').length;
    const atestados = baseSetor.filter((i) => i.statusAtual === 'ATESTADO').length;
    const outros = baseSetor.filter(
      (i) => i.statusAtual !== 'PRESENTE' && i.statusAtual !== 'FALTA' && i.statusAtual !== 'ATESTADO'
    ).length;

    return { total: baseSetor.length, presentes, faltas, atestados, outros };
  }, [listaCompletaTurno, setorFiltro]);

  // ==========================================
  // FUNÇÕES DE PERSONALIZAÇÃO DE FRENTES E SUB-FRENTES
  // ==========================================
  const renomearFrente = (setor, frenteId, novoNome) => {
    const nomeLimpo = novoNome.toUpperCase();
    const frenteAntiga = frentesConfig[setor].find((f) => f.id === frenteId);
    if (!frenteAntiga || !nomeLimpo.trim()) return;
    const nomeAntigo = frenteAntiga.nome;

    setFrentesConfig((prev) => ({
      ...prev,
      [setor]: prev[setor].map((f) => (f.id === frenteId ? { ...f, nome: nomeLimpo } : f))
    }));

    // Atualiza colaboradores que estavam naquela frente
    setFuncionarios((prev) =>
      prev.map((emp) =>
        emp.setor === setor && emp.frentePadrao === nomeAntigo
          ? { ...emp, frentePadrao: nomeLimpo }
          : emp
      )
    );

    setApontamentosPorDia((prev) => {
      const copia = { ...prev };
      Object.keys(copia).forEach((k) => {
        const diaObj = { ...copia[k] };
        Object.keys(diaObj).forEach((empId) => {
          if (diaObj[empId].frente === nomeAntigo) {
            diaObj[empId] = { ...diaObj[empId], frente: nomeLimpo };
          }
        });
        copia[k] = diaObj;
      });
      return copia;
    });
  };

  const alterarCorFrente = (setor, frenteId, novaCor) => {
    setFrentesConfig((prev) => ({
      ...prev,
      [setor]: prev[setor].map((f) => (f.id === frenteId ? { ...f, cor: novaCor } : f))
    }));
  };

  const alternarUsoSubfrentes = (setor, frenteId) => {
    setFrentesConfig((prev) => ({
      ...prev,
      [setor]: prev[setor].map((f) =>
        f.id === frenteId ? { ...f, temSubfrentes: !f.temSubfrentes } : f
      )
    }));
  };

  const adicionarNovaFrente = (e) => {
    e.preventDefault();
    if (!novaFrenteForm.nome.trim()) return;
    const nomeUpper = novaFrenteForm.nome.trim().toUpperCase();
    const nova = {
      id: `frente_${Date.now()}`,
      nome: nomeUpper,
      cor: novaFrenteForm.cor,
      temSubfrentes: novaFrenteForm.temSubfrentes,
      subfrentes: novaFrenteForm.temSubfrentes ? ['GERAL'] : []
    };

    setFrentesConfig((prev) => ({
      ...prev,
      [novaFrenteForm.setor]: [...prev[novaFrenteForm.setor], nova]
    }));
    setNovaFrenteForm({ ...novaFrenteForm, nome: '' });
    mostrarAviso(`Frente "${nomeUpper}" criada com sucesso!`);
  };

  const removerFrente = (setor, frenteId) => {
    if (frentesConfig[setor].length <= 1) {
      mostrarAviso('O setor precisa ter pelo menos uma frente ativa.');
      return;
    }
    setFrentesConfig((prev) => ({
      ...prev,
      [setor]: prev[setor].filter((f) => f.id !== frenteId)
    }));
    mostrarAviso('Frente removida!');
  };

  const adicionarSubfrente = (setor, frenteId) => {
    const texto = (novaSubfrenteInput[frenteId] || '').trim().toUpperCase();
    if (!texto) return;

    setFrentesConfig((prev) => ({
      ...prev,
      [setor]: prev[setor].map((f) =>
        f.id === frenteId && !f.subfrentes.includes(texto)
          ? { ...f, subfrentes: [...f.subfrentes, texto] }
          : f
      )
    }));
    setNovaSubfrenteInput((prev) => ({ ...prev, [frenteId]: '' }));
    mostrarAviso(`Sub-frente (Local) "${texto}" adicionada!`);
  };

  const removerSubfrente = (setor, frenteId, subNome) => {
    setFrentesConfig((prev) => ({
      ...prev,
      [setor]: prev[setor].map((f) =>
        f.id === frenteId
          ? { ...f, subfrentes: f.subfrentes.filter((s) => s !== subNome) }
          : f
      )
    }));
  };

  // ==========================================
  // EXPORTAÇÃO CSV IDÊNTICA AO ESQUELETO DO PDF
  // ==========================================
  const exportarEsqueletoCSV = () => {
    const { colunas, maxLinhas } = dadosEsqueletoPlanilha;
    const dataBR = dataSelecionada.split('-').reverse().join('/');
    let csv = '\uFEFF';
    csv += `"CONTROLE DE COLABORADORES";"${dataBR}"\n`;

    // Cabeçalho respeitando 2 colunas (FRENTE | ID) ou 3 colunas (DIVERSOS | ID | LOCAL)
    const cabecalho = colunas
      .map((col) =>
        col.temSubfrentes
          ? `"${col.nome}";"ID";"LOCAL"`
          : `"${col.nome}";"ID"`
      )
      .join(';');
    csv += cabecalho + '\n';

    for (let r = 0; r < maxLinhas; r++) {
      const linha = colunas
        .map((col) => {
          const emp = col.colaboradores[r];
          if (col.temSubfrentes) {
            return emp
              ? `"${emp.nome}";"${emp.id}";"${emp.subfrenteAtual || ''}"`
              : '"";"";""';
          }
          return emp ? `"${emp.nome}";"${emp.id}"` : '"";""';
        })
        .join(';');
      csv += linha + '\n';
    }

    const blob = new Blob([csv], { type: 'text/csv;charset=utf-8;' });
    const url = URL.createObjectURL(blob);
    const a = document.createElement('a');
    a.href = url;
    a.download = `CONTROLE_DE_FUNCIONARIOS_${dataSelecionada}_${setorFiltro}.csv`;
    a.click();
    mostrarAviso('Planilha exportada com o esqueleto idêntico ao PDF!');
  };

  // Copiar Manhã para Tarde
  const copiarManhaParaTarde = () => {
    const chaveManha = `${dataSelecionada}_MANHA`;
    const chaveTarde = `${dataSelecionada}_TARDE`;
    if (!apontamentosPorDia[chaveManha]) return;
    setApontamentosPorDia((prev) => ({
      ...prev,
      [chaveTarde]: JSON.parse(JSON.stringify(prev[chaveManha]))
    }));
    setTurnoSelecionado('TARDE');
    mostrarAviso('Apontamento da Manhã copiado para a Tarde!');
  };

  // Cadastrar Novo Colaborador
  const salvarNovoFuncionario = (e) => {
    e.preventDefault();
    if (!formNovoFunc.id.trim() || !formNovoFunc.nome.trim()) return;
    const fConfig = obterConfigFrente(formNovoFunc.setor, formNovoFunc.frentePadrao);
    const subPadrao =
      fConfig && fConfig.temSubfrentes
        ? formNovoFunc.subfrentePadrao || fConfig.subfrentes[0] || ''
        : '';

    const novo = {
      id: formNovoFunc.id.trim().toUpperCase(),
      nome: formNovoFunc.nome.trim().toUpperCase(),
      setor: formNovoFunc.setor,
      frentePadrao: formNovoFunc.frentePadrao,
      subfrentePadrao: subPadrao,
      responsavel: formNovoFunc.responsavel.trim().toUpperCase()
    };

    setFuncionarios((prev) => [novo, ...prev]);
    atualizarApontamento(novo.id, {
      status: 'PRESENTE',
      frente: novo.frentePadrao,
      subfrente: novo.subfrentePadrao,
      responsavel: novo.responsavel
    });

    setNovoFuncModal(false);
    setFormNovoFunc({
      id: '',
      nome: '',
      setor: setorFiltro,
      frentePadrao: frentesConfig[setorFiltro]?.[0]?.nome || '',
      subfrentePadrao: '',
      responsavel: setorFiltro
    });
    mostrarAviso(`Colaborador ${novo.nome} (${novo.id}) adicionado!`);
  };

  const dataFormatadaBR = dataSelecionada.split('-').reverse().join('/');
  const totalOcorrenciasTurno = kpis.faltas + kpis.atestados + kpis.outros;

  return (
    <div className="min-h-screen bg-slate-50 text-slate-800 font-sans pb-16 antialiased">
      {/* NOTIFICAÇÃO FLUTUANTE */}
      {notificacao && (
        <div className="fixed bottom-5 right-5 z-50 bg-slate-900 text-white px-4 py-3 rounded-2xl shadow-xl border border-slate-700 flex items-center gap-2.5 text-xs font-semibold animate-fade-in">
          <CheckCircle2 className="w-4 h-4 text-emerald-400 shrink-0" />
          <span>{notificacao}</span>
        </div>
      )}

      {/* CABEÇALHO UNIFICADO E MINIMALISTA */}
      <header className="bg-white border-b border-slate-200/80 sticky top-0 z-30 shadow-xs print:hidden">
        <div className="max-w-[1600px] mx-auto px-4 h-16 flex items-center justify-between gap-4">
          {/* Logo & Status de Salvamento */}
          <div className="flex items-center gap-3 shrink-0">
            <div className="bg-gradient-to-br from-amber-400 to-amber-500 text-slate-950 p-2 rounded-xl shadow-xs">
              <HardHat className="w-5 h-5" />
            </div>
            <div>
              <div className="flex items-center gap-2">
                <h1 className="text-sm sm:text-base font-extrabold text-slate-900 tracking-tight">
                  Controle de Obra
                </h1>
                <span
                  className="inline-flex items-center gap-1 px-2 py-0.5 rounded-full text-[10px] font-medium bg-emerald-50 text-emerald-700 border border-emerald-200/60"
                  title={ultimoSaveHora ? `Último salvamento às ${ultimoSaveHora}` : statusSave}
                >
                  <span className="w-1.5 h-1.5 rounded-full bg-emerald-500 animate-pulse" />
                  <span className="hidden md:inline">{statusSave}</span>
                  {ultimoSaveHora && <span className="text-emerald-600/80">• {ultimoSaveHora}</span>}
                </span>
              </div>
              <p className="text-[11px] text-slate-400 hidden sm:block">
                Gestão diária de frentes e efetivo
              </p>
            </div>
          </div>

          {/* Navegação Central em Pílulas Simplificadas */}
          <nav className="flex items-center gap-1 bg-slate-100/90 p-1 rounded-xl border border-slate-200/60 overflow-x-auto">
            {[
              { id: 'quadro', label: 'Planilha', icon: LayoutGrid },
              { id: 'celular', label: 'Celular', icon: Smartphone },
              { id: 'apontamento', label: 'Lista', icon: ClipboardList },
              {
                id: 'ocorrencias',
                label: 'Faltas',
                icon: AlertTriangle,
                badge: totalOcorrenciasTurno
              },
              { id: 'personalizar', label: 'Ajustes', icon: Palette }
            ].map((tab) => {
              const Icon = tab.icon;
              const ativo = abaAtiva === tab.id;
              return (
                <button
                  key={tab.id}
                  onClick={() => setAbaAtiva(tab.id)}
                  className={`flex items-center gap-1.5 px-3 py-1.5 rounded-lg text-xs font-bold transition whitespace-nowrap ${
                    ativo
                      ? 'bg-white text-slate-900 shadow-xs border border-slate-200/60'
                      : 'text-slate-500 hover:text-slate-900 hover:bg-white/50'
                  }`}
                >
                  <Icon className={`w-3.5 h-3.5 ${ativo ? 'text-amber-500' : 'text-slate-400'}`} />
                  <span>{tab.label}</span>
                  {tab.badge > 0 && (
                    <span
                      className={`px-1.5 py-0.2 rounded-full text-[10px] font-extrabold ${

                    <th className="py-3 px-4 font-bold">Local (Sub-frente)</th>
                    <th className="py-3 px-4 w-36 font-bold">Responsável</th>
                    <th className="py-3 px-4 font-bold">Status</th>
                    <th className="py-3 px-4 text-right w-24 font-bold">Ações</th>
                  </tr>
                </thead>
                <tbody className="divide-y divide-slate-100">
                  {listaFiltrada.map((emp) => {
                    const cfg = obterConfigFrente(emp.setor, emp.frenteAtual);
                    return (
                      <tr key={emp.id} className="hover:bg-slate-50/70 transition">
                        <td className="py-2 px-4">
                          <input
                            type="text"
                            defaultValue={emp.id}
                            key={`id_${emp.id}`}
                            onBlur={(e) => {
                              if (e.target.value.trim().toUpperCase() !== emp.id) {
                                const ok = alterarIdFuncionario(emp.id, e.target.value);
                                if (!ok) e.target.value = emp.id;
                              }
                            }}
                            onKeyDown={(e) => {
                              if (e.key === 'Enter') e.currentTarget.blur();
                            }}
                            className="w-20 font-mono font-bold text-xs bg-slate-100/80 border border-transparent hover:border-slate-300 focus:border-amber-500 focus:bg-white rounded-lg px-2 py-1 text-slate-800 uppercase focus:outline-none transition"
                          />
                        </td>

                        <td className="py-2 px-4">
                          <input
                            type="text"
                            value={emp.nome}
                            onChange={(e) =>
                              atualizarCadastroFuncionario(
                                emp.id,
                                { nome: e.target.value.toUpperCase() },
                                false
                              )
                            }
                            className="w-full font-semibold text-slate-900 bg-transparent border border-transparent hover:border-slate-200 focus:border-amber-500 focus:bg-white rounded-lg px-2 py-1 uppercase focus:outline-none transition"
                          />
                        </td>

                        <td className="py-2 px-4">
                          <select
                            value={emp.setor}
                            onChange={(e) => alterarSetorFuncionario(emp.id, e.target.value)}
                            className={`rounded-lg px-2.5 py-1 text-[11px] font-bold border ${
                              emp.setor === 'CIVIL'
                                ? 'bg-orange-50/70 text-orange-800 border-orange-200/70'
                                : 'bg-indigo-50/70 text-indigo-800 border-indigo-200/70'
                            }`}
                          >
                            <option value="METALICA">METÁLICA</option>
                            <option value="CIVIL">CIVIL</option>
                          </select>
                        </td>

                        <td className="py-2 px-4">
                          <select
                            value={emp.frenteAtual}
                            onChange={(e) => {
                              const novaF = e.target.value;
                              const novaCfg = obterConfigFrente(emp.setor, novaF);
                              const novaSub = novaCfg?.temSubfrentes
                                ? novaCfg.subfrentes[0] || ''
                                : '';
                              atualizarApontamento(emp.id, {
                                frente: novaF,
                                subfrente: novaSub
                              });
                            }}
                            className="border border-slate-200 rounded-lg px-2.5 py-1 font-semibold bg-white text-slate-800 focus:border-amber-500 focus:outline-none"
                          >
                            {(frentesConfig[emp.setor] || []).map((f) => (
                              <option key={f.id} value={f.nome}>
                                {f.nome}
                              </option>
                            ))}
                          </select>
                        </td>

                        <td className="py-2 px-4">
                          {cfg?.temSubfrentes ? (
                            <select
                              value={emp.subfrenteAtual}
                              onChange={(e) =>
                                atualizarApontamento(emp.id, { subfrente: e.target.value })
                              }
                              className="border border-purple-200 bg-purple-50/50 text-purple-900 rounded-lg px-2.5 py-1 font-semibold focus:outline-none"
                            >
                              {cfg.subfrentes.map((sub) => (
                                <option key={sub} value={sub}>
                                  {sub}
                                </option>
                              ))}
                            </select>
                          ) : (
                            <span className="text-slate-300">—</span>
                          )}
                        </td>

                        <td className="py-2 px-4">
                          <input
                            type="text"
                            value={emp.responsavelAtual || ''}
                            onChange={(e) =>
                              atualizarCadastroFuncionario(
                                emp.id,
                                { responsavel: e.target.value.toUpperCase() },
                                true
                              )
                            }
                            placeholder="—"
                            className="w-28 bg-transparent border border-transparent hover:border-slate-200 focus:bg-white focus:border-amber-500 rounded-lg px-2 py-1 text-xs font-medium text-slate-600 uppercase focus:outline-none transition"
                          />
                        </td>

                        <td className="py-2 px-4">
                          <select
                            value={emp.statusAtual}
                            onChange={(e) =>
                              atualizarApontamento(emp.id, { status: e.target.value })
                            }
                            className={`border rounded-lg px-2.5 py-1 font-bold text-[11px] focus:outline-none ${
                              emp.statusAtual === 'PRESENTE'
                                ? 'bg-emerald-50 text-emerald-800 border-emerald-200'
                                : emp.statusAtual === 'FALTA'
                                ? 'bg-rose-50 text-rose-800 border-rose-200'
                                : 'bg-amber-50 text-amber-800 border-amber-200'
                            }`}
                          >
                            {STATUS_OPCOES.map((st) => (
                              <option key={st} value={st} className="bg-white text-slate-800">
                                {st}
                              </option>
                            ))}
                          </select>
                        </td>

                        <td className="py-2 px-4 text-right whitespace-nowrap">
                          {confirmarExclusaoId === emp.id ? (
                            <div className="inline-flex items-center gap-1">
                              <button
                                type="button"
                                onClick={() => excluirFuncionario(emp.id)}
                                className="bg-rose-600 text-white px-2 py-1 rounded-md text-[10px] font-bold"
                              >
                                Sim
                              </button>
                              <button
                                type="button"
                                onClick={() => setConfirmarExclusaoId(null)}
                                className="bg-slate-200 text-slate-700 px-1.5 py-1 rounded-md text-[10px] font-bold"
                              >
                                ✕
                              </button>
                            </div>
                          ) : (
                            <div className="inline-flex items-center gap-1">
                              <button
                                type="button"
                                onClick={() => {
                                  setColabEditandoRapido(emp);
                                  setIdTempModal(emp.id);
                                  setConfirmarExclusaoId(null);
                                }}
                                className="p-1.5 text-slate-400 hover:text-slate-800 hover:bg-slate-100 rounded-lg transition"
                                title="Personalizar"
                              >
                                <Edit3 className="w-3.5 h-3.5" />
                              </button>
                              <button
                                type="button"
                                onClick={() => setConfirmarExclusaoId(emp.id)}
                                className="p-1.5 text-slate-400 hover:text-rose-600 hover:bg-rose-50 rounded-lg transition"
                                title="Excluir"
                              >
                                <Trash2 className="w-3.5 h-3.5" />
                              </button>
                            </div>
                          )}
                        </td>
                      </tr>
                    );
                  })}
                </tbody>
              </table>
            </div>
          </section>
        )}

        {/* =================================================================== */}
        {/* ABA 4: RELATÓRIO DE FALTAS E OCORRÊNCIAS DO DIA                     */}
        {/* =================================================================== */}
        {abaAtiva === 'ocorrencias' && (
          <section className="bg-white rounded-2xl shadow-xs border border-slate-200/90 overflow-hidden">
            <div className="p-5 border-b border-slate-200/80 flex flex-wrap items-center justify-between gap-3">
              <div>
                <h2 className="text-sm font-extrabold text-slate-900 uppercase">
                  Relatório de Faltas e Ocorrências — {dataFormatadaBR} ({setorFiltro})
                </h2>
                <p className="text-xs text-slate-500 mt-0.5">
                  Colaboradores ausentes, em atestado ou com batida não confirmada neste turno.
                </p>
              </div>
              <button
                onClick={gerarPdfSomenteFaltas}
                disabled={gerandoPdf}
                className="flex items-center gap-1.5 bg-rose-600 hover:bg-rose-500 text-white px-4 py-2 rounded-xl text-xs font-bold shadow-xs transition"
              >
                {gerandoPdf ? (
                  <Loader2 className="w-4 h-4 animate-spin" />
                ) : (
                  <FileDown className="w-4 h-4" />
                )}
                Baixar PDF de Faltas
              </button>
            </div>

            <table className="w-full text-left border-collapse text-xs">
              <thead>
                <tr className="bg-slate-50/80 text-slate-500 text-[11px] uppercase border-b border-slate-200">
                  <th className="py-3 px-5 font-bold">Funcionário</th>
                  <th className="py-3 px-4 font-bold w-28 text-center">ID</th>
                  <th className="py-3 px-4 font-bold w-48 text-center">Responsável</th>
                  <th className="py-3 px-5 font-bold w-80">Justificativa / Status</th>
                </tr>
              </thead>
              <tbody className="divide-y divide-slate-100">
                {listaCompletaTurno
                  .filter((i) => i.setor === setorFiltro && i.statusAtual !== 'PRESENTE')
                  .map((oc) => (
                    <tr key={oc.id} className="hover:bg-slate-50/70 transition">
                      <td className="py-3 px-5 font-bold text-slate-900">{oc.nome}</td>
                      <td className="py-3 px-4 font-mono font-bold text-center text-slate-600">
                        {oc.id}
                      </td>
                      <td className="py-3 px-4 text-center">
                        <input
                          type="text"
                          value={oc.responsavelAtual}
                          onChange={(e) =>
                            atualizarApontamento(oc.id, {
                              responsavel: e.target.value.toUpperCase()
                            })
                          }
                          className="border border-slate-200 rounded-lg px-2.5 py-1 text-center font-semibold w-36 focus:border-amber-500 focus:outline-none"
                        />
                      </td>
                      <td className="py-3 px-5 flex items-center justify-between gap-2">
                        <span className="px-2.5 py-1 rounded-full bg-rose-50 text-rose-700 border border-rose-200 font-bold text-[11px]">
                          {oc.statusAtual}
                        </span>
                        <button
                          onClick={() => atualizarApontamento(oc.id, { status: 'PRESENTE' })}
                          className="text-emerald-600 hover:text-emerald-700 font-bold text-[11px] hover:underline"
                        >
                          Marcar Presente
                        </button>
                      </td>
                    </tr>
                  ))}
                {listaCompletaTurno.filter(
                  (i) => i.setor === setorFiltro && i.statusAtual !== 'PRESENTE'
                ).length === 0 && (
                  <tr>
                    <td colSpan={4} className="p-10 text-center text-slate-400">
                      <CheckCircle2 className="w-8 h-8 text-emerald-400 mx-auto mb-2 opacity-80" />
                      <p className="font-bold text-slate-600 text-xs">
                        100% do efetivo presente!
                      </p>
                      <p className="text-[11px] text-slate-400 mt-0.5">
                        Nenhuma falta ou ocorrência registrada neste turno para {setorFiltro}.
                      </p>
                    </td>
                  </tr>
                )}
              </tbody>
            </table>
          </section>
        )}

        {/* =================================================================== */}
        {/* ABA 5: PERSONALIZAR FRENTES, CORES & FUNCIONÁRIOS EM TEMPO REAL     */}
        {/* =================================================================== */}
        {abaAtiva === 'personalizar' && (
          <section className="space-y-5">
            {/* Seletor Limpo entre Frentes e Funcionários */}
            <div className="bg-white p-3 rounded-2xl shadow-xs border border-slate-200/90 flex flex-wrap items-center justify-between gap-3">
              <div className="inline-flex bg-slate-100 p-1 rounded-xl border border-slate-200/60">
                <button
                  type="button"
                  onClick={() => setSubAbaPersonalizar('frentes')}
                  className={`flex items-center gap-2 px-4 py-2 rounded-lg text-xs font-bold transition ${
                    subAbaPersonalizar === 'frentes'
                      ? 'bg-white text-slate-900 shadow-xs'
                      : 'text-slate-600 hover:text-slate-900'
                  }`}
                >
                  <Palette className="w-4 h-4 text-amber-500" />
                  Frentes, Cores & Locais ({setorFiltro})
                </button>
                <button
                  type="button"
                  onClick={() => setSubAbaPersonalizar('funcionarios')}
                  className={`flex items-center gap-2 px-4 py-2 rounded-lg text-xs font-bold transition ${
                    subAbaPersonalizar === 'funcionarios'
                      ? 'bg-white text-slate-900 shadow-xs'
                      : 'text-slate-600 hover:text-slate-900'
                  }`}
                >
                  <Users className="w-4 h-4 text-sky-500" />
                  Cadastro de Funcionários ({funcionarios.filter((f) => f.setor === setorFiltro).length})
                </button>
              </div>

              <button
                type="button"
                onClick={() => {
                  setFormNovoFunc({
                    id: '',
                    nome: '',
                    setor: setorFiltro,
                    frentePadrao: frentesConfig[setorFiltro]?.[0]?.nome || '',
                    subfrentePadrao: '',
                    responsavel: setorFiltro
                  });
                  setNovoFuncModal(true);
                }}
                className="flex items-center gap-1.5 bg-slate-900 hover:bg-slate-800 text-white px-4 py-2 rounded-xl text-xs font-bold shadow-xs transition"
              >
                <UserPlus className="w-4 h-4 text-amber-400" />
                Cadastrar Funcionário
              </button>
            </div>

            {subAbaPersonalizar === 'frentes' ? (
              <div className="space-y-5">
                {/* Criar Nova Frente */}
                <div className="bg-white p-5 rounded-2xl shadow-xs border border-slate-200/90">
                  <h3 className="text-xs font-extrabold text-slate-900 uppercase tracking-wider flex items-center gap-2 mb-3">
                    <Plus className="w-4 h-4 text-amber-500" />
                    Criar Nova Frente em {setorFiltro}
                  </h3>
                  <form
                    onSubmit={adicionarNovaFrente}
                    className="grid grid-cols-1 md:grid-cols-4 gap-3 items-end text-xs"
                  >
                    <div>
                      <label className="block font-semibold text-slate-600 mb-1">Nome da Frente</label>
                      <input
                        type="text"
                        required
                        value={novaFrenteForm.nome}
                        onChange={(e) =>
                          setNovaFrenteForm({
                            ...novaFrenteForm,
                            setor: setorFiltro,
                            nome: e.target.value
                          })
                        }
                        placeholder="Ex: COBERTURA, PINTURA..."
                        className="w-full border border-slate-200 bg-slate-50 focus:bg-white rounded-xl px-3 py-2 font-bold uppercase focus:border-amber-500 focus:outline-none"
                      />
                    </div>

                    <div>
                      <label className="block font-semibold text-slate-600 mb-1">Cor da Coluna</label>
                      <div className="flex items-center gap-2">
                        <input
                          type="color"
                          value={novaFrenteForm.cor}
                          onChange={(e) =>
                            setNovaFrenteForm({ ...novaFrenteForm, cor: e.target.value })
                          }
                          className="w-10 h-9 rounded-lg cursor-pointer border border-slate-200"
                        />
                        <span className="font-mono text-slate-500">{novaFrenteForm.cor}</span>
                      </div>
                    </div>

                    <div className="flex items-center gap-2 pb-2">
                      <input
                        type="checkbox"
                        id="chkSub"
                        checked={novaFrenteForm.temSubfrentes}
                        onChange={(e) =>
                          setNovaFrenteForm({
                            ...novaFrenteForm,
                            temSubfrentes: e.target.checked
                          })
                        }
                        className="w-4 h-4 rounded"
                      />
                      <label htmlFor="chkSub" className="font-semibold text-slate-700 cursor-pointer">
                        Usar Sub-frentes (Coluna LOCAL)
                      </label>
                    </div>

                    <button
                      type="submit"
                      className="bg-slate-900 hover:bg-slate-800 text-white font-bold py-2.5 px-4 rounded-xl shadow-xs transition"
                    >
                      + Adicionar Frente
                    </button>
                  </form>
                </div>

                {/* Cards de Cada Frente Existente */}
                <div className="grid grid-cols-1 md:grid-cols-2 lg:grid-cols-3 gap-4">
                  {frentesDoSetorAtivo.map((frente) => (
                    <div
                      key={frente.id}
                      className="bg-white rounded-2xl shadow-xs border border-slate-200/90 overflow-hidden flex flex-col"
                    >
                      <div
                        style={{ backgroundColor: frente.cor }}
                        className="px-4 py-3 text-white flex items-center justify-between"
                      >
                        <span className="font-extrabold text-xs tracking-wider uppercase">
                          {frente.nome}
                        </span>
                        <button
                          onClick={() => removerFrente(setorFiltro, frente.id)}
                          className="text-white/80 hover:text-white text-[11px] flex items-center gap-1 bg-black/20 hover:bg-black/30 px-2.5 py-1 rounded-lg transition"
                        >
                          <Trash2 className="w-3 h-3" /> Excluir
                        </button>
                      </div>

                      <div className="p-4 space-y-4 text-xs flex-1">
                        <div>
                          <label className="block font-semibold text-slate-500 mb-1">
                            Nome da Frente
                          </label>
                          <input
                            type="text"
                            value={frente.nome}
                            onChange={(e) =>
                              renomearFrente(setorFiltro, frente.id, e.target.value)
                            }
                            className="w-full border border-slate-200 bg-slate-50 focus:bg-white rounded-xl px-3 py-2 font-bold uppercase text-slate-900 focus:border-amber-500 focus:outline-none transition"
                          />
                        </div>

                        <div>
                          <label className="block font-semibold text-slate-500 mb-1.5">
                            Cor na Planilha
                          </label>
                          <div className="flex flex-wrap items-center gap-1.5">
                            {CORES_RAPIDAS.map((c) => (
                              <button
                                key={c}
                                type="button"
                                onClick={() => alterarCorFrente(setorFiltro, frente.id, c)}
                                style={{ backgroundColor: c }}
                                className={`w-6 h-6 rounded-full border-2 transition ${
                                  frente.cor === c
                                    ? 'border-amber-400 scale-110 shadow-xs'
                                    : 'border-transparent opacity-85 hover:opacity-100'
                                }`}
                              />
                            ))}
                            <input
                              type="color"
                              value={frente.cor}
                              onChange={(e) =>
                                alterarCorFrente(setorFiltro, frente.id, e.target.value)
                              }
                              className="w-6 h-6 rounded-full cursor-pointer border border-slate-300 ml-1"
                              title="Escolher outra cor"
                            />
                          </div>
                        </div>

                        <div className="pt-3 border-t border-slate-100 space-y-2.5">
                          <div className="flex items-center justify-between">
                            <label className="font-bold text-slate-700 flex items-center gap-1.5">
                              <Layers className="w-3.5 h-3.5 text-purple-600" />
                              Sub-frentes (LOCAL)
                            </label>
                            <button
                              type="button"
                              onClick={() => alternarUsoSubfrentes(setorFiltro, frente.id)}
                              className={`px-2.5 py-0.5 rounded-full font-bold text-[10px] transition ${
                                frente.temSubfrentes
                                  ? 'bg-purple-600 text-white'
                                  : 'bg-slate-100 text-slate-600'
                              }`}
                            >
                              {frente.temSubfrentes ? 'Ativo' : 'Desativado'}
                            </button>
                          </div>

                          {frente.temSubfrentes && (
                            <div className="bg-purple-50/50 p-3 rounded-xl border border-purple-200/70 space-y-2">
                              <div className="flex gap-1.5">
                                <input
                                  type="text"
                                  value={novaSubfrenteInput[frente.id] || ''}
                                  onChange={(e) =>
                                    setNovaSubfrenteInput({
                                      ...novaSubfrenteInput,
                                      [frente.id]: e.target.value
                                    })
                                  }
                                  placeholder="Novo local..."
                                  className="flex-1 bg-white border border-purple-200 rounded-lg px-2.5 py-1.5 uppercase font-semibold focus:outline-none"
                                />
                                <button
                                  type="button"
                                  onClick={() => adicionarSubfrente(setorFiltro, frente.id)}
                                  className="bg-purple-700 hover:bg-purple-600 text-white px-3 py-1.5 rounded-lg font-bold"
                                >
                                  +
                                </button>
                              </div>

                              <div className="flex flex-wrap gap-1 max-h-36 overflow-y-auto pt-1">
                                {frente.subfrentes.map((sub) => (
                                  <span
                                    key={sub}
                                    className="inline-flex items-center gap-1 bg-white border border-purple-200 text-purple-950 px-2 py-0.5 rounded-md font-semibold text-[10px]"
                                  >
                                    {sub}
                                    <button
                                      type="button"
                                      onClick={() =>
                                        removerSubfrente(setorFiltro, frente.id, sub)
                                      }
                                      className="text-rose-500 hover:text-rose-700 font-bold ml-0.5"
                                    >
                                      ×
                                    </button>
                                  </span>
                                ))}
                              </div>
                            </div>
                          )}
                        </div>
                      </div>
                    </div>
                  ))}
                </div>
              </div>
            ) : (
              /* SUB-ABA: PERSONALIZAR FUNCIONÁRIOS EM TEMPO REAL */
              <div className="bg-white rounded-2xl shadow-xs border border-slate-200/90 overflow-hidden">
                <div className="p-4 border-b border-slate-200/80 flex flex-col md:flex-row md:items-center justify-between gap-3">
                  <div>
                    <h3 className="text-sm font-extrabold text-slate-900">
                      Base de Funcionários ({setorFiltro})
                    </h3>
                    <p className="text-xs text-slate-500">
                      Edite qualquer campo abaixo para atualizar a alocação padrão e a planilha em tempo real.
                    </p>
                  </div>

                  <div className="flex items-center gap-2 bg-slate-50 px-3 py-2 rounded-xl border border-slate-200 w-full md:w-80">
                    <Search className="w-4 h-4 text-slate-400" />
                    <input
                      type="text"
                      value={busca}
                      onChange={(e) => setBusca(e.target.value)}
                      placeholder="Buscar por nome, ID ou frente..."
                      className="bg-transparent text-xs text-slate-800 w-full focus:outline-none"
                    />
                    {busca && (
                      <button
                        type="button"
                        onClick={() => setBusca('')}
                        className="text-slate-400 hover:text-slate-600"
                      >
                        <X className="w-4 h-4" />
                      </button>
                    )}
                  </div>
                </div>

                <div className="overflow-x-auto max-h-[600px]">
                  <table className="w-full text-left border-collapse text-xs">
                    <thead className="bg-slate-50 text-slate-500 text-[11px] uppercase sticky top-0 z-10 border-b border-slate-200">
                      <tr>
                        <th className="py-3 px-4 w-28 font-bold">ID</th>
                        <th className="py-3 px-4 min-w-[240px] font-bold">Nome Completo</th>
                        <th className="py-3 px-4 w-32 font-bold">Setor</th>
                        <th className="py-3 px-4 font-bold">Frente Padrão</th>
                        <th className="py-3 px-4 font-bold">Local Padrão</th>
                        <th className="py-3 px-4 w-36 font-bold">Responsável</th>
                        <th className="py-3 px-4 text-right w-24 font-bold">Excluir</th>
                      </tr>
                    </thead>
                    <tbody className="divide-y divide-slate-100">

                                <input
                                  type="text"

                                      emp.id,

                              </td>

                                  value={emp.setor}

                                <input
                                  type="text"

            <div className="bg-slate-900 text-white px-5 py-4 flex items-center justify-between">
              <h3 className="font-bold text-sm flex items-center gap-2">

                      setFormNovoFunc({

              <button
                onClick={gerarPdfSomenteFaltas}
                disabled={gerandoPdf}
                className="w-full text-left p-3.5 rounded-xl border border-rose-200 bg-rose-50/40 hover:bg-rose-100/50 transition flex items-center justify-between gap-3"
              >
                <div>
                  <span className="font-black text-rose-950 text-sm block">
                    4. Somente Relatório de Faltas e Ocorrências (A4 Retrato)
                  </span>
                  <span className="text-rose-700 text-[11px]">
                    Lista apenas os colaboradores ausentes, com atestado ou batida não encontrada no dia.
                  </span>
                </div>
                <span className="bg-rose-600 text-white font-black px-3 py-1.5 rounded-lg shrink-0">
                  Faltas
                </span>
              </button>
            </div>

            <div className="bg-slate-100 px-5 py-3 flex items-center justify-between border-t border-slate-200">
              <span className="text-[11px] text-slate-500">
                {gerandoPdf ? 'Gerando documento vetorial...' : 'Pronto para download imediato'}
              </span>
              <button
                onClick={() => setModalPdfAberto(false)}
                className="px-4 py-1.5 rounded-lg border border-slate-300 bg-white text-slate-700 font-bold text-xs hover:bg-slate-50"
              >
                Fechar
              </button>
            </div>
          </div>
        </div>
      )}

      {/* =================================================================== */}
      {/* MODAL PARA REINICIAR PLANILHA DO DIA ATUAL                          */}
      {/* =================================================================== */}
      {modalReiniciarDia && (
        <div className="fixed inset-0 z-50 bg-slate-950/60 backdrop-blur-sm flex items-center justify-center p-4">
          <div className="bg-white rounded-2xl shadow-2xl border border-slate-200 max-w-md w-full overflow-hidden">
            <div className="bg-slate-900 text-white px-5 py-4 flex items-center justify-between">
              <div className="flex items-center gap-2">
                <RotateCcw className="w-4 h-4 text-amber-400" />
                <h3 className="font-black text-sm">
                  Reiniciar Planilha de {dataFormatadaBR} ({turnoSelecionado})
                </h3>
              </div>
              <button
                onClick={() => setModalReiniciarDia(false)}
                className="text-slate-400 hover:text-white"
              >
                <X className="w-5 h-5" />
              </button>
            </div>

            <div className="p-5 space-y-3 text-xs">
              <p className="text-slate-600 leading-relaxed">
                Esta ação reinicia <strong>apenas o dia {dataFormatadaBR} ({turnoSelecionado})</strong>. Todas as planilhas de dias anteriores continuarão salvas normalmente no histórico.
              </p>

              <button
                type="button"
                onClick={() => reiniciarPlanilhaDiaAtual('PADRAO')}
                className="w-full text-left p-3.5 rounded-xl border-2 border-emerald-500 bg-emerald-50/60 hover:bg-emerald-100/70 transition"
              >
                <span className="font-black text-emerald-950 text-sm block">
                  1. Restaurar Alocação Padrão (Todos Presentes)
                </span>
                <span className="text-emerald-800 text-[11px]">
                  Todos os colaboradores voltam como PRESENTE nas suas frentes e sub-frentes padrão.
                </span>
              </button>

              <button
                type="button"
                onClick={() => reiniciarPlanilhaDiaAtual('EM_BRANCO')}
                className="w-full text-left p-3.5 rounded-xl border border-slate-300 hover:bg-slate-50 transition"
              >
                <span className="font-black text-slate-900 text-sm block">
                  2. Iniciar Chamada do Zero (Todos Não Confirmados)
                </span>
                <span className="text-slate-600 text-[11px]">
                  Deixa a planilha limpa para que você marque um por um quem está presente na obra hoje.
                </span>
              </button>

              <div className="pt-2 flex justify-end">
                <button
                  type="button"
                  onClick={() => setModalReiniciarDia(false)}
                  className="px-4 py-2 rounded-lg border border-slate-300 text-slate-700 font-bold hover:bg-slate-50"
                >
                  Cancelar
                </button>
              </div>
            </div>
          </div>
        </div>
      )}
    </div>
  );
}
